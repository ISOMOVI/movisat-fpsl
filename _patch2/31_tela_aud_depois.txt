    // morto". `sem_registro` é o terceiro estado, e não é atraso.
    let passada = 'Última passada: sem registro ainda.';
    if (!d.sem_registro) {
      const q = new Date(d.ultima_em);
      const quando = isNaN(q) ? d.ultima_em

        <strong>${d.idade_maxima_cache_h}h</strong>: dado velho acusaria problema que não existe.
        <br>${passada}`);
