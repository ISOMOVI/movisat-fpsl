// 🚨 "LIGADA" NUNCA FOI "RODOU" (16/10). Até aqui a tela só dizia o estado do
// interruptor; se o laço morresse, ela continuaria dizendo "Rotina ligada" com
// a mesma cara. Agora a nota carrega a ÚLTIMA PASSADA, e os três estados são
// distintos: carimbo recente, carimbo velho (suspeita de atraso) e sem
// registro -- este último não é atraso, é serviço que acabou de subir.
function passadaEmTexto(d) {
  if (d.sem_registro) return 'Última passada: sem registro ainda.';
  const q = new Date(d.ultima_em);
  const quando = isNaN(q) ? d.ultima_em
    : q.toLocaleString('pt-BR', { day: '2-digit', month: '2-digit', hour: '2-digit', minute: '2-digit' });
  const resultado = d.ultimo_resultado ? ` — ${d.ultimo_resultado}` : '';
  return (d.atrasada ? 'Última passada (ATRASADA): ' : 'Última passada: ') + quando + resultado + '.';
}

function pintarChave(d) {
  document.getElementById('chave').checked = d.ativa;
  document.getElementById('chaveTx').textContent =
    d.ativa ? 'Rotina ligada' : 'Rotina desligada';
  document.getElementById('chaveNota').textContent = (d.ativa
    ? `Roda sozinha a cada ${d.intervalo_min} min — ligar não roda na hora.`
    : (ehDono ? 'Nada roda sozinho enquanto estiver assim.'
              : 'Nada roda sozinho enquanto estiver assim. Só o proprietário liga.'))
    + ' ' + passadaEmTexto(d);
